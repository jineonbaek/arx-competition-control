cmake_minimum_required(VERSION 3.16)
project(arx_waypoint_rviz)
find_package(ament_cmake REQUIRED)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(deps rviz_common rviz_default_plugins pluginlib rclcpp geometry_msgs std_msgs)
foreach(dep IN LISTS deps)
  find_package(${dep} REQUIRED)
endforeach()
add_library(${PROJECT_NAME} SHARED waypoints.cpp foreground_polygon.cpp startup_view.cpp)
ament_target_dependencies(${PROJECT_NAME} ${deps})
target_link_libraries(${PROJECT_NAME} Qt5::Widgets)
pluginlib_export_plugin_description_file(rviz_common plugins.xml)
add_executable(test_point_inputs test_point_inputs.cpp)
target_compile_features(test_point_inputs PUBLIC cxx_std_17)
target_link_libraries(test_point_inputs Qt5::Widgets)
add_executable(test_startup_view test_startup_view.cpp)
target_compile_features(test_startup_view PUBLIC cxx_std_17)
add_executable(test_route_list test_route_list.cpp)
target_compile_features(test_route_list PUBLIC cxx_std_17)
target_link_libraries(test_route_list Qt5::Widgets)
enable_testing()
add_test(NAME route_list COMMAND test_route_list)
set_tests_properties(route_list PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
add_test(NAME startup_view COMMAND test_startup_view)
add_test(NAME point_inputs COMMAND test_point_inputs)
set_tests_properties(point_inputs PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION lib)
ament_package()
