cmake_minimum_required(VERSION 3.16)
project(arx_base_mask_costmap)
find_package(ament_cmake REQUIRED)
find_package(nav2_costmap_2d REQUIRED)
find_package(pluginlib REQUIRED)
find_package(ament_index_cpp REQUIRED)
find_package(tf2_geometry_msgs REQUIRED)
find_package(nav2_util REQUIRED)
add_library(base_mask_layers SHARED layers.cpp)
target_compile_features(base_mask_layers PUBLIC cxx_std_17)
ament_target_dependencies(base_mask_layers nav2_costmap_2d pluginlib ament_index_cpp tf2_geometry_msgs)
target_link_libraries(base_mask_layers geos_c)
pluginlib_export_plugin_description_file(nav2_costmap_2d plugin.xml)
install(TARGETS base_mask_layers LIBRARY DESTINATION lib)
install(FILES base_envelope.wkt DESTINATION share/${PROJECT_NAME})
add_executable(test_base_geometry test_geometry.cpp)
target_compile_features(test_base_geometry PUBLIC cxx_std_17)
target_link_libraries(test_base_geometry geos_c)
enable_testing()
add_test(NAME base_geometry COMMAND test_base_geometry)
foreach(target test_native_baseline test_base_layers)
  add_executable(${target} test_layers.cpp)
  target_compile_features(${target} PUBLIC cxx_std_17)
  ament_target_dependencies(${target} nav2_costmap_2d pluginlib ament_index_cpp tf2_geometry_msgs nav2_util)
  target_link_libraries(${target} geos_c)
endforeach()
target_compile_definitions(test_native_baseline PRIVATE BASELINE)
add_test(NAME base_layers COMMAND test_base_layers ${CMAKE_CURRENT_SOURCE_DIR}/base_envelope.wkt)
set_tests_properties(base_layers PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=198")
add_executable(test_base_loading test_loading.cpp)
target_compile_features(test_base_loading PUBLIC cxx_std_17)
ament_target_dependencies(test_base_loading nav2_costmap_2d pluginlib)
add_test(NAME base_loading COMMAND test_base_loading)
set_tests_properties(base_loading PROPERTIES ENVIRONMENT "ROS_DOMAIN_ID=198")
ament_package()
